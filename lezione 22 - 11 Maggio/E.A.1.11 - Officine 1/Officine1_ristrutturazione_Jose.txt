3.  Eliminazione delle generalizzazioni tra classi
	[V.Staff.dipendente_assunzione]
		ALL s:
		Staff(s) -> 
		(	tipo(s,Dipendente) 
			<-> 
			EXISTS dip_assunzione: assunzione(s,dip_assunzione)
		)
	[V.Staff.direttore_nascita]
		ALL s:
		Staf(s)->
		(	tipo(s, Direttore)
			<->
			EXISTS dir_nascita: nascita(s, dir_nascita)
		)
		
	[V.Officina.diretta_da_direttore]
		ALL dir:
		Staff(dir) AND Officina(off) AND dirige(off,dir)
		->
		tipo(dir,Direttore)
	[V.RiparazioneTerminata.riconsegna_dopo_accettazione]
		ALL r,x,y:
		RiparazioneTerminata(r) AND accettazione(r,x) AND
		riconsegna(r,y)
		->
		y>x
	[V.Riparazione.isa.disjoint]
		ALL r,ric,rt:
		Riparazione(r) AND RiparazioneInCorso(ric) AND
		RiparazioneTerminata(rt)
		->
		NOT ( rt_isa_r(rt,r) AND ric_isa_r(ric,r) )
		
	[V.Riparazione.isa.complete]
		ALL r:
		Riparazione(r)
		->
		EXISTS x: ric_isa_r(x,r) OR rt_isa_r(x,r)
